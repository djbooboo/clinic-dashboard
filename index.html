/** * ==================================================== * 파일명: Code.gs (다정부부한의원 업무 대시보드 백엔드) * 작성일시: 2026-09-12 21:25:00 * 주요 기능: * 1. Status 시트 출퇴근 데이터 조회 및 저장 * 2. Notices 시트 실시간 임시 공지 저장 및 불러오기 * 3. Calendar 시트 근무/업무/스페셜 이벤트 정규화 저장 및 조회 * ==================================================== */ function doGet(e) { // [작성일시: 2026-09-12 21:25:00] 데이터 조회 및 웹 앱 JSON 반환 var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Status 시트 읽기 var statusSheet = ss.getSheetByName("Status") || ss.insertSheet("Status"); var statusRows = statusSheet.getDataRange().getValues(); var statuses = {}; for (var i = statusRows.length - 1; i >= 1; i--) { var name = String(statusRows[i][1]).trim(); var timeStr = String(statusRows[i][2]).trim(); if (name && !statuses[name]) { if (timeStr.indexOf('~') !== -1 && timeStr.split('~')[1].trim() !== '') { statuses[name] = 'OUT'; } else { statuses[name] = 'IN'; } } } // 2. Notices 시트 읽기 var noticeSheet = ss.getSheetByName("Notices") || ss.insertSheet("Notices"); var noticeVal = noticeSheet.getRange(1, 1).getValue(); var notices = []; try { if (noticeVal) notices = JSON.parse(noticeVal); } catch(err) { notices = []; } // 3. Calendar 시트 읽기 (스페셜 이벤트 열 추가 반영 - [작성일시: 2026-09-12 21:25:00]) var calSheet = ss.getSheetByName("Calendar") || ss.insertSheet("Calendar"); var calDisplayRows = calSheet.getDataRange().getDisplayValues(); var calRows = calSheet.getDataRange().getValues(); var calendar = {}; for (var j = 0; j < calRows.length; j++) { var rawDateStr = calDisplayRows[j][0] ? String(calDisplayRows[j][0]).trim() : ""; var dateKey = ""; if (calRows[j][0] instanceof Date) { dateKey = Utilities.formatDate(calRows[j][0], Session.getScriptTimeZone(), "yyyy-MM-dd"); } else { dateKey = rawDateStr; } if (dateKey && dateKey !== '일자Key' && dateKey !== '날짜Key' && dateKey !== '일자') { calendar[dateKey] = { morningWorker: calRows[j][1] ? String(calRows[j][1]) : "", afternoonWorker: calRows[j][2] ? String(calRows[j][2]) : "", worker: calRows[j][1] ? String(calRows[j][1]) : "", task: calRows[j][3] ? String(calRows[j][3]) : "", isHoliday: (calRows[j][4] === true || String(calRows[j][4]).toUpperCase() === 'TRUE'), isHalfDay: (calRows[j][5] === true || String(calRows[j][5]).toUpperCase() === 'TRUE'), specialEvent: calRows[j][6] ? String(calRows[j][6]) : "" }; } } var output = { statuses: statuses, notices: notices, calendar: calendar }; return ContentService.createTextOutput(JSON.stringify(output)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { // [작성일시: 2026-09-12 21:25:00] 데이터 입력 및 시트 저장 처리 try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. 출퇴근 데이터 기록 if (data.name && data.type) { var statusSheet = ss.getSheetByName("Status") || ss.insertSheet("Status"); if (statusSheet.getLastRow() === 0) { statusSheet.appendRow(["일자", "이름", "출퇴근시간", "근무시간"]); } var now = new Date(); var days = ['일', '월', '화', '수', '목', '금', '토']; var month = String(now.getMonth() + 1).padStart(2, '0'); var date = String(now.getDate()).padStart(2, '0'); var dateStr = month + '/' + date + ' (' + days[now.getDay()] + ')'; var timeNow = String(now.getHours()).padStart(2, '0') + ':' + String(now.getMinutes()).padStart(2, '0'); if (data.type === '출근') { statusSheet.appendRow([dateStr, data.name, timeNow + ' ~ ', '']); } else if (data.type === '퇴근') { var rows = statusSheet.getDataRange().getValues(); var targetRowIndex = -1; var inTimeStr = ''; for (var i = rows.length - 1; i >= 1; i--) { if (rows[i][1] === data.name && String(rows[i][2]).indexOf('~') !== -1) { var parts = String(rows[i][2]).split('~'); if (parts[1].trim() === '') { targetRowIndex = i + 1; inTimeStr = parts[0].trim(); break; } } } if (targetRowIndex > 0) { var fullTimeStr = inTimeStr + ' ~ ' + timeNow; var workHours = ''; try { var inParts = inTimeStr.split(':'); var inDate = new Date(); inDate.setHours(parseInt(inParts[0]), parseInt(inParts[1]), 0); var diffMs = now.getTime() - inDate.getTime(); if (diffMs > 0) workHours = (diffMs / (1000 * 60 * 60)).toFixed(2); } catch(err) {} statusSheet.getRange(targetRowIndex, 3).setValue(fullTimeStr); statusSheet.getRange(targetRowIndex, 4).setValue(workHours); } else { statusSheet.appendRow([dateStr, data.name, '~ ' + timeNow, '']); } } } // 2. 임시 공지 사항 저장 if (data.type === 'UPDATE_NOTICES') { var noticeSheet = ss.getSheetByName("Notices") || ss.insertSheet("Notices"); noticeSheet.clear(); noticeSheet.getRange(1, 1).setValue(JSON.stringify(data.notices)); } // 3. 달력 저장 (스페셜 이벤트 포함 7개 열 저장 - [작성일시: 2026-09-12 21:25:00]) if (data.type === 'UPDATE_CALENDAR') { var calSheet = ss.getSheetByName("Calendar") || ss.insertSheet("Calendar"); var calDisplayRows = calSheet.getDataRange().getDisplayValues(); var calRows = calSheet.getDataRange().getValues(); var targetRow = -1; for (var k = 0; k < calRows.length; k++) { var dateStr = ""; if (calRows[k][0] instanceof Date) { dateStr = Utilities.formatDate(calRows[k][0], Session.getScriptTimeZone(), "yyyy-MM-dd"); } else { dateStr = calDisplayRows[k][0] ? String(calDisplayRows[k][0]).trim() : ""; } if (dateStr === data.dateKey) { targetRow = k + 1; break; } } var morn = data.morningWorker || ""; var aft = data.afternoonWorker || ""; var task = data.task || ""; var isHol = data.isHoliday === true; var isHalf = data.isHalfDay === true; var specialEvent = data.specialEvent || ""; var dateKeyText = "'" + data.dateKey; if (targetRow > 0) { calSheet.getRange(targetRow, 1, 1, 7).setValues([[dateKeyText, morn, aft, task, isHol, isHalf, specialEvent]]); } else { calSheet.appendRow([dateKeyText, morn, aft, task, isHol, isHalf, specialEvent]); } } return ContentService.createTextOutput(JSON.stringify({status: "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
